"""Read retained evidence and check the proposed document's links and claims."""
import hashlib
import json
from pathlib import Path
import re

REPO = Path('C:/Users/prince.baah/Downloads/autortl-work')
STAGE = Path(__file__).resolve().parent
DOC = STAGE / 'docs/CASE_STUDY_VERIFIER.md'
OLD = REPO / 'evidence/2026-10-26/signed-fir4-generated-audit-01'
NEW = REPO / 'evidence/2026-10-26/fir4-driver-comparison-01'
SIGNED = REPO / 'evidence/2026-10-26/fir4-signed-boundary-comparison-01'
ADMIT = REPO / 'evidence/2026-10-25/signed-fir4-admission-attempt-01'
REGRESSION = REPO / 'evidence/2026-10-26/fir4-coverage-tests-01'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads(p.read_text(encoding='utf-8'))
old = read(OLD / 'audit/result.json')
new = read(NEW / 'comparison/result.json')
signed = read(SIGNED / 'comparison/result.json')
admit = read(ADMIT / 'admission/result.json')
regression = read(REGRESSION / 'result.json')
assert (old['detected'], old['missed'], old['ungraded']) == (18, 9, 0)
assert (new['detected'], new['missed'], new['ungraded']) == (22, 5, 0)
assert old['control_passed'] and new['reference_passed']
assert admit['qualified'] and admit['planned'] == admit['executed'] == admit['admitted'] == 28
assert len(admit['results']) == 28
assert admit['results'][0]['name'] == 'reference' and admit['results'][0]['simulation_passed']
assert all(row['admitted'] and row['assertion_proof'] and not row['simulation_passed']
           for row in admit['results'][1:])
names = {row['name'] for row in old['results']}
assert len(names) == 28 and names == {row['name'] for row in new['audit']} == {row['name'] for row in admit['results']}
assert sum(row['grade'] == 'detected' for row in old['results']) == 18
assert sum(row['grade'] == 'missed' for row in old['results']) == 9
assert sum(row['grade'] == 'detected' for row in new['audit']) == 22
assert sum(row['grade'] == 'missed' for row in new['audit']) == 5
gains = set(new['detected_names']) - set(old['detected_names'])
assert gains == {'truncate12', 'history_on_invalid', 'async_history_reset', 'async_valid_reset'}
assert set(new['missed_names']) == {'stale_history_reset', 'retained_stage1_reset',
                                  'retained_stage2_reset', 'retained_stage3_reset', 'combinational_reset_gate'}

# These are the actual tested source bytes, not merely matching mutant names.
for name in names:
    assert sha(OLD / 'audit' / name / 'children.v') == sha(NEW / 'comparison/audit' / name / 'children.v')
    assert sha(OLD / 'audit' / name / 'children.v') == sha(ADMIT / 'admission' / name / 'children.v')

exit_groups = [(old['results'], OLD / 'audit'), (new['audit'], NEW / 'comparison/audit'),
               (admit['results'], ADMIT / 'admission'), (new['controls'], NEW / 'comparison/controls'),
               ([dict(new['producer_probe'], name='producer_probe')], NEW / 'comparison')]
verified_cases = 0
for rows, directory in exit_groups:
    for row in rows:
        exits = row['process_exits']
        case = directory / row['name']
        assert exits['schema_valid'] and row['inputs_unchanged']
        assert sha(case / 'tool_exit_records.log') == exits['sha256']
        raw = (case / 'sim_log.txt').read_text(encoding='utf-8')
        if row.get('grade') == 'detected':
            assert re.search(r'(?m)^\s*FAIL\b', raw) and 'TEST FAILED' in raw and 'TEST PASSED' not in raw
        if row.get('grade') in ('missed', 'control_passed'):
            assert 'TEST PASSED' in raw and 'TEST FAILED' not in raw
        for tool in ('xvlog', 'xelab', 'xsim'):
            latest = exits['latest_by_tool'][tool]
            assert latest['successful'] and latest['returncode'] == latest['driver_returncode'] == 0
            assert latest['status'] == 'exited'
            assert sha(case / Path(latest['output_log_path']).name) == latest['output_log_sha256']
        verified_cases += 1

oracle_old = REPO / 'evidence/2026-10-26/generated-signed-fir4-attempt-01/run/modules/signed_fir4_core/golden.py'
assert sha(oracle_old) == sha(SIGNED / 'comparison/golden.py') == sha(NEW / 'comparison/golden.py')
assert (SIGNED / 'comparison/generated/stimulus.json').read_bytes() == (NEW / 'comparison/generated/stimulus.json').read_bytes()
assert signed['passed'] and signed['old_reference_passed'] and signed['new_reference_passed']
assert signed['old_truncate12_missed'] and signed['new_truncate12_detected'] and signed['same_golden_reference_mutant']
assert (signed['old_payloads'], signed['new_payloads']) == (104, 110)
assert (signed['old_expected_min'], signed['old_expected_max']) == (-1703, 1539)
assert (signed['new_expected_min'], signed['new_expected_max']) == (-2169, 2166)
assert signed['original_prefix_and_random_input_tail_preserved']
assert all(row['passed'] for row in new['controls']) and len(new['controls']) == 5
prolonged = next(row for row in new['controls'] if row['name'] == 'prolonged_block')
assert prolonged['coverage']['counts']['longest_block'] == 16
assert prolonged['coverage']['counts']['producer_errors'] == 0
assert new['producer_probe']['bad_driver_detected']
assert new['producer_probe']['coverage']['counts']['producer_errors'] == 9
assert regression['passed'] == 1450 and regression['exit_code'] == 0 and regression['source_unchanged']
assert old['model_calls'] == old['synthesis_calls'] == new['model_calls'] == new['synthesis_calls'] == 0

text = DOC.read_text(encoding='utf-8')
# Resolve as if installed in repository docs/, not under this isolated stage.
links = []
for target in re.findall(r'\[[^\]]*\]\(([^)]+)\)', text):
    path = ((REPO / 'docs') / target.split('#', 1)[0]).resolve()
    assert path.is_relative_to(REPO.resolve()) and path.is_file(), target
    links.append({'target': target, 'repo_path': path.relative_to(REPO).as_posix(), 'sha256': sha(path)})
word_count = len(text.split())
assert word_count <= 700, word_count
result = {
    'document': 'docs/CASE_STUDY_VERIFIER.md', 'document_sha256': sha(DOC), 'words_whitespace_count': word_count,
    'links_checked': len(links), 'links': links,
    'measured': {'original': [18, 27], 'corrected': [22, 27], 'independent': [27, 27],
                 'remaining_misses': sorted(new['missed_names']), 'newly_detected': sorted(gains),
                 'same_RTL_cases_byte_verified': len(names), 'cases_with_verified_zero_tool_exits_and_logs': verified_cases,
                 'same_golden_sha256': sha(oracle_old),
                 'signed_only_and_scheduling_payload_arrays_same_sha256': sha(NEW / 'comparison/generated/stimulus.json'),
                 'nominal_controls': 5, 'producer_probe_errors': 9, 'historical_software_passes': 1450},
    'source_mapping': {
        'historical_driver_patch': {'path': str((NEW / 'stage-review/tbgen.patch').relative_to(REPO)),
                                    'sha256': sha(NEW / 'stage-review/tbgen.patch')},
        'historical_mutations': {'path': str((ADMIT / 'fixture/qualification/cases.py').relative_to(REPO)),
                                'sha256': sha(ADMIT / 'fixture/qualification/cases.py')},
        'historical_reference': {'path': str((ADMIT / 'fixture/qualification/reference.v').relative_to(REPO)),
                                'sha256': sha(ADMIT / 'fixture/qualification/reference.v')},
        'current_maintained_driver': {'path': 'autortl/tbgen.py', 'sha256': sha(REPO / 'autortl/tbgen.py'),
                                      'limit': 'Current source includes later changes; historical patch is used for the measured attribution.'}
    },
    'scope': 'Read-only artifact checks and proposed documentation only. No model/vendor/test/board jobs or canonical edits. Historical source/result evidence, not a fresh simulation or formal proof.'
}
(STAGE / 'linkcheck-and-source-map.json').write_text(json.dumps(result, indent=2) + '\n', encoding='utf-8')
print(json.dumps({k: v for k, v in result.items() if k not in ('links', 'source_mapping')}, indent=2))
