def gf_multiply(a, b):
    product = 0
    for bit in range(8):
        if b & 1:
            product = product ^ a
        high = a & 128
        a = (a << 1) & 255
        if high:
            a = a ^ 27
        b = b >> 1
    return product


def gf_power(value, exponent):
    result = 1
    while exponent:
        if exponent & 1:
            result = gf_multiply(result, value)
        value = gf_multiply(value, value)
        exponent = exponent >> 1
    return result


def init():
    sbox = []
    for value in range(256):
        inverse = 0
        if value:
            inverse = gf_power(value, 254)
        result = inverse
        for shift in range(1, 5):
            rotated = ((inverse << shift) | (inverse >> (8 - shift))) & 255
            result = result ^ rotated
        sbox.append(result ^ 99)
    return dict(sbox=sbox)


def step(state, inputs):
    sbox = state.get("sbox", [])
    state_in = int(inputs.get("state_in", 0)) & ((1 << 128) - 1)
    round_key = int(inputs.get("round_key", 0)) & ((1 << 128) - 1)
    final_round = int(inputs.get("final_round", 0)) & 1

    substituted = []
    for index in range(16):
        shift = 120 - 8 * index
        substituted.append(sbox[(state_in >> shift) & 255])

    shifted = [0] * 16
    for row in range(4):
        for column in range(4):
            source_column = (column + row) % 4
            shifted[row + 4 * column] = substituted[row + 4 * source_column]

    result_bytes = shifted
    if not final_round:
        mixed = [0] * 16
        for column in range(4):
            a0 = shifted[4 * column]
            a1 = shifted[1 + 4 * column]
            a2 = shifted[2 + 4 * column]
            a3 = shifted[3 + 4 * column]

            d0 = ((a0 << 1) & 255) ^ (27 if a0 & 128 else 0)
            d1 = ((a1 << 1) & 255) ^ (27 if a1 & 128 else 0)
            d2 = ((a2 << 1) & 255) ^ (27 if a2 & 128 else 0)
            d3 = ((a3 << 1) & 255) ^ (27 if a3 & 128 else 0)

            mixed[4 * column] = d0 ^ d1 ^ a1 ^ a2 ^ a3
            mixed[1 + 4 * column] = a0 ^ d1 ^ d2 ^ a2 ^ a3
            mixed[2 + 4 * column] = a0 ^ a1 ^ d2 ^ d3 ^ a3
            mixed[3 + 4 * column] = d0 ^ a0 ^ a1 ^ a2 ^ d3
        result_bytes = mixed

    result = 0
    for index in range(16):
        result = (result << 8) | result_bytes[index]

    outputs = dict(state_out=(result ^ round_key) & ((1 << 128) - 1))
    return state, outputs
