import json,subprocess,sys
from pathlib import Path
work=Path(__file__).resolve().parent
rows=[]
for name,old,new in [
    ('nov8_benchmark_review_probe','nov8-benchmark-review-probes','nov8-benchmark-review-probes-fixed'),
    ('nov8_benchmark_inventory_probe','nov8-benchmark-inventory-probe','nov8-benchmark-inventory-probe-fixed'),
]:
    path=work/(name+'_fixed.py')
    path.write_text((work/(name+'.py')).read_text(encoding='utf-8').replace(old,new),encoding='utf-8')
    result=subprocess.run([sys.executable,str(path)],capture_output=True,text=True,encoding='utf-8')
    assert result.returncode==0,result.stderr
    print(result.stdout)
    rows.append({'probe':name,'exit_code':result.returncode})
cases=json.loads((work/'nov8-benchmark-review-probes-fixed/results.json').read_text())['cases']
assert cases[0]['outcome']=='accepted'
assert all(row['outcome'].startswith('ArtifactError:') for row in cases[1:]),cases
inventory=json.loads((work/'nov8-benchmark-inventory-probe-fixed/result.json').read_text())
assert inventory['outcome'].startswith('ArtifactError:'),inventory
(work/'nov8-benchmark-postfix-probes.json').write_text(json.dumps({'exit_code':0,'probes':rows,'all_original_findings_resolved':True},indent=2)+'\n',encoding='utf-8')
